\begin{restatable}{lemma}{expanderFwalk} \label{thm:Din06Prop2.5}
	% serve without self-loops? No, purché un cappio abbia contributo 2 al grado
	Sia $G = (V, E)$ un $(d, \lambda)$-expander e $F \subseteq E$ un sottoinsieme
	degli archi. Sia $K$ la distribuzione di probabilità sui vertici indotta
	dalla scelta di un estremo di un arco in $F$ scelto uniformemente a caso.
	La probabilità $p$ che una passeggiata aleatoria che comincia con
	distribuzione $K$ faccia l'$(i+1)$-esimo passo in $F$ è maggiorata da
	$\frac{|F|}{|E|} + \lambda^i$.
\end{restatable}

\begin{proof} % [Dimostrazione del Lemma \ref{thm:Din06Prop2.5}]
	Siano $x$ il vettore distribuzione di $K$, cioè	$x_v = \frac{\deg_F(v)}
	{2 |F|}$, e $y_v$ la probabilità che un arco incidente a $v$ sia in $F$, cioè
	$y = \frac{2 |F|}{d} x$.
	Scomponiamo $x = x^\parallel + x^\perp$ con
	$x^\parallel := \frac1n \mathbf{1}$. Vale per definizione
	\[
		p = \sum_{v \in V}{ y_v (A^i x)_v }
		  = \langle y, A^i x \rangle.
	\]

	Osserviamo $\lVert x \rVert ^2 \le \big(\sum_v{\lvert x_v \rvert} \big)
	\max_v{\lvert x_v \rvert} = \max_v{\lvert x_v \rvert}
	\le \frac{d}{2 | F |}$ da cui, applicando la disuguaglianza di Cauchy-Schwarz
	e la definizione variazionale di $\lambda$, otteniamo
	\[
		\big\langle y, A^i x^\perp \big\rangle
		\le \lVert y \rVert \cdot \lVert A^i x^\perp \rVert
		\le \frac{2 |F|}{d} \lVert x \rVert \cdot \lambda^i \lVert x \rVert
		\le \lambda^i.
	\]

	In conclusione,
	\[
		\langle y, A^i x \rangle
		= \big\langle y, A^i x^\parallel \big\rangle
		+ \big\langle y, A^i x^\perp \big\rangle
		\le \frac{2 |F|}{n d} + \lambda^i.
	\]
\end{proof}

\begin{lemma} \label{thm:ESbarS}
	Sia $G = (V, E)$ un $(n, d, \lambda)$-expander e $S \subseteq V$ un
	sottoinsieme dei vertici. Allora
	\[
		E(S, \bar{S}) \ge (1 - \lambda)\frac{d}{n}|S||\bar{S}|.
	\]
\end{lemma}
\begin{proof}
	Esprimiamo $E(S, \bar S)$ come un opportuno prodotto scalare e analizziamo
	separatamente componente parallela e perpendicolare al vettore costante.
	Detta $A$ la matrice di adiacenza rinormalizzata, siano $\mathbf{1}_S$ e
	$\mathbf{1}_{\bar{S}}$ i vettori indicatori di $S$ e del suo complementare.
	Siano anche $\mathbf{1}_S^\parallel = \frac{|S|}{n} \mathbf{1}$ e
	$\mathbf{1}_S^\perp = \mathbf{1}_S - \mathbf{1}_S^\parallel$ le componenti
	di $\mathbf{1}_S$ parallela e perpendicolare a $\mathbf{1}$.
	Vale
	\begin{align*}
		\frac{1}{d} E(S, \bar{S}) &= \langle \mathbf{1}_S, A \mathbf{1}_{\bar S} \rangle \\
		&= \langle \mathbf{1}_S, A (\mathbf{1} - \mathbf{1}_{\bar S}) \rangle \\
		&= \langle \mathbf{1}_S, \mathbf{1} \rangle
			- \langle \mathbf{1}_S, A \mathbf{1}_S \rangle \\
		&= |S| - \big\langle \mathbf{1}_S^\parallel + \mathbf{1}_S^\perp,
			A (\mathbf{1}_S^\parallel + \mathbf{1}_S^\perp ) \big\rangle \\
		&= |S| - \big\langle \mathbf{1}_S^\parallel, \mathbf{1}_S^\parallel \big\rangle
			- \big\langle \mathbf{1}_S^\perp, A \mathbf{1}_S^\perp \big\rangle
			- 2 \big\langle \mathbf{1}_S^\parallel, \mathbf{1}_S^\perp \big\rangle \\
		&\ge |S| - \big\langle \mathbf{1}_S^\parallel, \mathbf{1}_S^\parallel \big\rangle
			- \lambda \big\langle \mathbf{1}_S^\perp, \mathbf{1}_S^\perp \big\rangle \\
		&= |S| - \frac{|S|^2}{n} - \lambda\bigg(|S| - \frac{|S|^2}{n}\bigg) \\
		&= (1 - \lambda)\frac{|S|}{n}(n - |S|) = (1 - \lambda) \frac1{n} |S| |\bar S|.
	\end{align*}
\end{proof}
