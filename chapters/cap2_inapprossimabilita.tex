\chapter{Teorema PCP e Inapprossimabilità}

\section{Inapprossimabilità di Max-Clique}

\begin{theorem}[Feige, Goldwasser, Lovasz, Safra, Szegedy, `91] \label{thm:max-clique}
	Per ogni $\varepsilon > 0$, non esiste una $\varepsilon$-approssimazione di Max-Clique, a meno che $\textbf{P} = \textbf{NP}$.
\end{theorem}
